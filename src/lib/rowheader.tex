% \rowtimeloc{location}{time}
% Standalone right-aligned time + location block.
\newcommand{\rowtimeloc}[3][0.5\linewidth]{%
  \begin{minipage}[t]{#1}
    \raggedleft
    {#2}\\[1pt]
    {#3}
  \end{minipage}%
}

% \rowheader{title}{subtitle}{location}{time}
\newcommand{\rowheader}[4]{%
  \noindent
  \begin{minipage}[t]{0.5\linewidth}
    \raggedright
    \textbf{#1}\\[1pt]
    \textit{#2}
  \end{minipage}%
  \hfill%
  \rowtimeloc{#3}{#4}%
  \par\vspace{2pt}%
}
