% Job data system + dependencies
\usepackage{etoolbox}
\usepackage{xstring}
\usepackage{needspace}

% --- Experience row renderer ---
\newcommand{\expitem}[2]{%
  \par\noindent
  \makebox[1em][l]{\textbullet}%
  \parbox[t]{\dimexpr\linewidth-1em}{\textbf{#1:}~#2}\par\vspace{2pt}%
}

\newcommand{\experiencerow}[7]{%
  \needspace{6\baselineskip}%
  \rowheader{#1}{#2}{#3}{#4 -- #5}%
  \nopagebreak[4]%
  \noindent\hspace{1em}\colorbox{gray!12}{%
    \parbox{\dimexpr\linewidth-1em-2\fboxsep\relax}{%
      \begin{list}{}{%
          \setlength{\leftmargin}{2.2em}%
          \setlength{\itemindent}{-1.8em}%
          \setlength{\topsep}{0pt}%
          \setlength{\itemsep}{0pt}%
          \setlength{\parsep}{0pt}%
        }%
      \item\texttt{\small \$ \{#6\}}%
      \end{list}%
    }%
  }\par%
  \nopagebreak[4]%
  \vspace{3pt}%
  \sloppy #7%
  \vspace{6pt}%
}

% --- Job data system ---
\newcommand{\definejob}[9]{%
  \expandafter\providecommand\csname JOB-#1-#2\endcsname{}%
  \expandafter\def\csname JOB-#1-#2\endcsname{%
    \experiencerow{#3}{#4}{#5}{#6}{#7}{#8}{#9}%
  }%
  \expandafter\providecommand\csname JOBLIST-#1\endcsname{}%
  \expandafter\xdef\csname JOBLIST-#1\endcsname{\csuse{JOBLIST-#1},#2}%
}

\newcommand{\renderjob}[2]{%
  \csname JOB-#1-#2\endcsname%
}

\newcommand{\renderjobs}[1]{%
  \def\ns{#1}%
  \def\dorender##1{\renderjob{\ns}{##1}}%
  \edef\thejoblist{\csuse{JOBLIST-#1}}%
  \expandafter\forcsvlist\expandafter\dorender\expandafter{\thejoblist}%
}
