% Job data system + dependencies
\usepackage{etoolbox}
\usepackage{xstring}
\usepackage{needspace}

% --- Experience row renderer ---
\newcommand{\expitem}[2]{%
  \noindent\hangindent=1.1em\hangafter=1%
  \textbullet~\textbf{#1:}~#2\par%
}

\newcommand{\experiencerow}[7]{%
  % #1=Company, #2=Title, #3=Location, #4=Start, #5=End, #6=Tech, #7=bullets
  \needspace{6\baselineskip}%
  \noindent
  \begin{minipage}[t]{0.5\linewidth}
    \raggedright
    \textbf{#1}\\[1pt]
    \textit{#2}
  \end{minipage}%
  \hfill%
  \begin{minipage}[t]{0.5\linewidth}
    \raggedleft
    {#3}\\[1pt]
    {#4 -- #5}
  \end{minipage}%
  \nopagebreak[4]%
  \par\vspace{2pt}%
  \noindent\texttt{\small #6}\par%
  \nopagebreak[4]%
  \vspace{3pt}%
  \sloppy #7%
  \vspace{6pt}%
}

% --- Job data system ---
\newcommand{\alljobs}{}

\newcommand{\definejob}[8]{%
  \ifx\alljobs\empty
    \edef\alljobs{#1}%
  \else
    \edef\alljobs{\alljobs,#1}%
  \fi
  \expandafter\newcommand\csname job@#1\endcsname{%
    \experiencerow{#2}{#3}{#4}{#5}{#6}{#7}{#8}%
  }%
}

\newcommand{\renderjob}[1]{%
  \csname job@#1\endcsname%
}

\newcommand{\renderjobs}[1]{%
  \edef\joblist{#1}%
  \expandafter\forcsvlist\expandafter\renderjob\expandafter{\joblist}%
}
